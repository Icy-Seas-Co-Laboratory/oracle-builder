# macOS GPU install profile.
# TensorFlow's official pip page does not list official macOS GPU support,
# but Apple provides tensorflow-metal as a PluggableDevice for Mac GPUs.
-r requirements.txt
# tensorflow-metal 1.2.0 documents compatibility through the TensorFlow 2.18 line.
tensorflow>=2.18,<2.19; platform_system == "Darwin"
tensorflow-metal==1.2.0; platform_system == "Darwin"
# Keep image/GUI transitive dependencies compatible with TensorFlow 2.18's NumPy pin.
numpy>=2.0,<2.1; platform_system == "Darwin"
tifffile<2026.5.0; platform_system == "Darwin"
